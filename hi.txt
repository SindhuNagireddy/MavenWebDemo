
// code for when object create lambda triiger
//dynamodbtable name:newtable3 (Object-String)
import boto3
table = boto3.resource('dynamodb').Table('newtable3')
def lambda_handler(event, context):
    print("TRIGGERED")
    for r in event['Records']:
        table.put_item(Item={
            'Object': r['s3']['object']['key']
        })
